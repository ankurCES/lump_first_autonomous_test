7/// Unique identifier for a job.
8pub type JobId = u64;
9
10/// Lifecycle status of a job.
11#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
12pub enum Status {
13    Open,
14    Done,
15    Cancelled,
16}
17
18/// Scheduling priority of a job.
19#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
20pub enum Priority {
21    High,
22    Med,
23    Low,
24}
25
26/// A single unit of work tracked by dispatchd.
27#[derive(Debug, Clone, Serialize, Deserialize)]
28pub struct Job {
29    pub id: JobId,
30    pub title: String,
31    pub priority: Priority,
32    pub status: Status,
33    pub created_at: DateTime<Utc>,
34}
35
36/// Persistent job queue backed by a JSON file.
37pub struct Store {
38    jobs: Vec<Job>,
39    next_id: JobId,
40    path: PathBuf,
41}
42
43/// Internal envelope used for JSON serialisation.
44#[derive(Serialize, Deserialize)]
45struct StoreData {
46    jobs: Vec<Job>,
47    next_id: JobId,
48}
49
50impl Store {
51    /// Returns the path to the jobs file.
52    ///
53    /// Uses `$DISPATCHD_HOME/jobs.json` when the env var is set, otherwise
54    /// falls back to `~/.dispatchd/jobs.json`.
55    pub fn default_path() -> PathBuf {
56        if let Ok(home) = std::env::var("DISPATCHD_HOME") {
57            PathBuf::from(home).join("jobs.json")
58        } else {
59            let home = dirs_home();
60            home.join(".dispatchd").join("jobs.json")
61        }
62    }
63
64    /// Loads the store from [`default_path`](Self::default_path).
65    ///
66    /// Returns an empty store when the file does not exist yet.
67    pub fn load() -> anyhow::Result<Store> {
68        let path = Self::default_path();
69        if !path.exists() {
70            return Ok(Store { jobs: Vec::new(), next_id: 1, path });
71        }
72        let file = std::fs::File::open(&path)
73            .with_context(|| format!("opening {}", path.display()))?;
74        let data: StoreData = serde_json::from_reader(file)
75            .with_context(|| format!("parsing {}", path.display()))?;
76        Ok(Store { jobs: data.jobs, next_id: data.next_id, path })
77    }
78
79    /// Persists the store to disk using an atomic tmp+rename write.
80    ///
81    /// Writes to `<path>.tmp` first (same directory = same filesystem partition),
82    /// then renames to the final path. On success no `.tmp` file remains.
83    pub fn save(&self) -> anyhow::Result<()> {
84        if let Some(parent) = self.path.parent() {
85            std::fs::create_dir_all(parent)
86                .with_context(|| format!("creating dir {}", parent.display()))?;
87        }
88        let data = StoreData { jobs: self.jobs.clone(), next_id: self.next_id };
89        let json = serde_json::to_string_pretty(&data)?;
90
91        // Write to a .tmp file in the SAME directory (same filesystem → rename is atomic)
92        let tmp_path = self.path.with_extension("json.tmp");
93        std::fs::write(&tmp_path, &json)
94            .with_context(|| format!("writing tmp file {}", tmp_path.display()))?;
95        std::fs::rename(&tmp_path, &self.path)
96            .with_context(|| format!("renaming {} to {}", tmp_path.display(), self.path.display()))?;
97        Ok(())
98    }
99
100    /// Adds a new job and returns its assigned [`JobId`].
101    pub fn add(&mut self, title: String, priority: Priority) -> JobId {
102        let id = self.next_id;
103        self.next_id += 1;
104        self.jobs.push(Job {
105            id,
106            title,
107            priority,
108            status: Status::Open,
109            created_at: Utc::now(),
110        });
111        id
112    }
113
114    /// Returns jobs matching `filter`, or all jobs when `filter` is `None`.
115    pub fn list(&self, filter: Option<Status>) -> Vec<&Job> {
116        self.jobs
117            .iter()
118            .filter(|j| match &filter {
119                None => true,
120                Some(s) => &j.status == s,
121            })
122            .collect()
123    }
124
125    /// Marks the job with the given `id` as [`Status::Cancelled`].
126    ///
127    /// Returns an error if no job with that id exists.
128    pub fn cancel(&mut self, id: JobId) -> anyhow::Result<()> {
129        let job = self.jobs.iter_mut().find(|j| j.id == id)
130            .ok_or_else(|| anyhow!("job {id} not found"))?;
131        job.status = Status::Cancelled;
132        Ok(())
133    }
134
135    /// Marks the job with the given `id` as [`Status::Done`].
136    ///
137    /// Returns an error if no job with that id exists.
138    pub fn complete(&mut self, id: JobId) -> anyhow::Result<()> {
139        let job = self.jobs.iter_mut().find(|j| j.id == id)
140            .ok_or_else(|| anyhow!("job {id} not found"))?;
141        job.status = Status::Done;
142        Ok(())
143    }
144}
145
146/// Resolves `$HOME` without pulling in an extra crate.
147fn dirs_home() -> PathBuf {
148    std::env::var("HOME")
149        .map(PathBuf::from)
150        .unwrap_or_else(|_| PathBuf::from("."))
151}

153#[cfg(test)]

156    use tempfile::TempDir;

158    fn store_in_tmpdir(dir: &TempDir) -> Store {
159        let path = dir.path().join("jobs.json");
160        // Build a Store with a custom path (not default_path)
161        Store { jobs: Vec::new(), next_id: 1, path }
162    }
163
164    #[test]
165    fn test_round_trip() {
166        let dir = TempDir::new().unwrap();
167        let mut store = store_in_tmpdir(&dir);
168        store.add("alpha".into(), Priority::High);
169        store.add("beta".into(), Priority::Med);
170        store.add("gamma".into(), Priority::Low);
171        store.save().unwrap();
172
173        // Reload and verify
174        let path = dir.path().join("jobs.json");
175        let file = std::fs::File::open(&path).expect("jobs.json must exist after save");
176        let data: StoreData = serde_json::from_reader(file).expect("jobs.json must be valid JSON");
177        let reloaded = Store { jobs: data.jobs, next_id: data.next_id, path };
178
179        let jobs = reloaded.list(None);
180        assert_eq!(jobs.len(), 3, "expected 3 jobs after reload");
181        assert_eq!(jobs[0].title, "alpha");
182        assert_eq!(jobs[1].title, "beta");
183        assert_eq!(jobs[2].title, "gamma");
184        assert_eq!(jobs[0].id, 1);
185        assert_eq!(jobs[1].id, 2);
186        assert_eq!(jobs[2].id, 3);
187    }
188
189    #[test]
190    fn test_atomic_rename() {
191        let dir = TempDir::new().unwrap();
192        let mut store = store_in_tmpdir(&dir);
193        store.add("task".into(), Priority::Low);
194        store.save().unwrap();
195
196        // No .tmp file should remain
197        let tmp = dir.path().join("jobs.json.tmp");
198        assert!(!tmp.exists(), ".tmp file must not exist after save");
199    }

201    #[test]
202    fn test_monotonic_id() {
203        let dir = TempDir::new().unwrap();
204        let mut store = store_in_tmpdir(&dir);
205        let id1 = store.add("first".into(), Priority::High);
206        let id2 = store.add("second".into(), Priority::Med);
207        let id3 = store.add("third".into(), Priority::Low);
208        assert!(id1 < id2, "ids must increase: {} < {}", id1, id2);
209        assert!(id2 < id3, "ids must increase: {} < {}", id2, id3);
